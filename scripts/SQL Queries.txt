--query to find top 5 customers by total sales
SELECT "Customer Name", sum(Sales) as Totalsales FROM Superstores GROUP by "Customer Name" ORDER by Totalsales DESC limit 5;

--query for top 5 products by quantity sold
SELECT "Product Name",Category, "Sub-Category", sum(Quantity) as Quantitysold, sum(Sales) as Totalsales from Superstores GROUP by "Product Name",Category,"Sub-Category" ORDER by Quantitysold DESC LIMIT 5;

--query for low-profit products (profit margin < 10%)
SELECT "Product Name",Category, (sum(profit)/sum(Sales)) as profitmargin FROM Superstores GROUP by "Product Name",Category HAVING profitmargin<0.1 ORDER by profitmargin DESC;
